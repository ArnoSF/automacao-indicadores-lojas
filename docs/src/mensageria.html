 37def conectar_servidor_smtp(
 38    email_remetente: str, senha_email: str, porta_smtp: str, servidor_smtp: str
 39) -> smtplib.SMTP | None:
 40    """Abre conexão segura (TLS) e realiza autenticação no servidor SMTP.
 41
 42    Args:
 43        email_remetente (str): Endereço de e-mail da conta remetente.
 44        senha_email (str): Senha ou senha de aplicativo da conta remetente.
 45        porta_smtp (str): Porta de comunicação do servidor SMTP.
 46        servidor_smtp (str): Endereço do host SMTP.
 47
 48    Returns:
 49        smtplib.SMTP | None: Objeto de conexão SMTP autenticado, ou None em caso de falha.
 50    """
 51    try:
 52        servidor_email = smtplib.SMTP(servidor_smtp, int(porta_smtp))
 53        servidor_email.starttls()
 54        servidor_email.login(email_remetente, senha_email)
 55        return servidor_email
 56    except Exception as erro:
 57        registrar_log(f"Erro na conexão com o servidor SMTP: {erro}!", "CRITICAL")
 58        return None

 75def _converter_df_bytes(dic_abas: dict[str, pd.DataFrame]) -> bytes:
 76    """Converte um dicionário de DataFrames em bytes de abas formatadas no formato Excel (.xlsx).
 77
 78    Args:
 79        dic_abas (dict[str, pd.DataFrame]): Dicionário mapeando {nome_aba: dataframe}.
 80
 81    Returns:
 82        bytes: Conteúdo binário do arquivo Excel gerado.
 83    """
 84    buffer = io.BytesIO()
 85
 86    formatos_diretoria = {
 87        "Faturamento (R$)": "#,##0.00",
 88        "Meta Faturamento (R$)": "#,##0.00",
 89        "Diversidade de Produtos": "#,##0",
 90        "Meta Diversidade": "#,##0",
 91        "Ticket Médio (R$)": "#,##0.00",
 92        "Meta Ticket (R$)": "#,##0.00",
 93    }
 94    semaforos_diretoria = ["Faturamento Ok", "Diversidade Ok", "Ticket Ok"]
 95
 96    with pd.ExcelWriter(buffer, engine="openpyxl") as writer:
 97        for aba, df in dic_abas.items():
 98            df.to_excel(writer, index=False, sheet_name=aba)
 99            _formatar_aba_excel(
100                writer.sheets[aba],
101                df,
102                formatos_colunas=formatos_diretoria,
103                colunas_semaforo=semaforos_diretoria,
104            )
105
106    return buffer.getvalue()

109def enviar_email(
110    servidor_email: smtplib.SMTP,
111    assunto: str,
112    remetente: str,
113    destinatario: str,
114    html_corpo: str,
115    dic_imagens_embutidas: dict[str, bytes],
116    caminho_planilha: Path | None,
117    bytes_anexo: bytes,
118    nome_anexo: str,
119    manter_planilhas: bool,
120    exibir_prints: bool,
121    id_log: str,
122) -> bool:
123    """Envia e-mails com corpo HTML, imagens embutidas e planilha Excel em anexo.
124
125    Args:
126        servidor_email (smtplib.SMTP): Servidor de e-mail a ser utilizado para o envio da mensagem.
127        assunto (str): Assunto da mensagem.
128        remetente (str): Endereço de e-mail do remetente da mensagem.
129        destinatario (str): Endereço de e-mail do destinatário da mensagem.
130        html_corpo (str): HTML do corpo da mensagem.
131        dic_imagens_embutidas (dict[str, bytes]): Dicionário mapeando {cid_com_colchetes: img_bytes}.
132        caminho_planilha (Path | None): Caminho da planilha de loja a ser removida, se configurado.
133        bytes_anexo (bytes): Arquivo em bytes para ser anexado na mensagem.
134        nome_anexo (str): Nome do arquivo que será anexado na mensagem.
135        manter_planilhas (bool): Indicador de manutenção da planilha em armazenamento local.
136        exibir_prints (bool): Indicador para exibição de mensagens informativas durante a execução.
137        id_log (str): Texto identificador a ser registrado no arquivo de log em caso de erro.
138
139    Returns:
140        bool: Sucesso ou não da operação.
141    """
142    try:
143        msg = EmailMessage()
144        msg["Subject"] = assunto
145        msg["From"] = remetente
146        msg["To"] = destinatario
147
148        msg.set_content("Seu leitor de e-mail não suporta visualização em HTML.")
149        msg.add_alternative(html_corpo, subtype="html")
150
151        for cid_com_colch, img_bytes in dic_imagens_embutidas.items():
152            msg.get_payload()[1].add_related(img_bytes, maintype="image", subtype="png", cid=cid_com_colch)
153
154        msg.add_attachment(
155            bytes_anexo,
156            maintype="application",
157            subtype="vnd.openxmlformats-officedocument.spreadsheetml.sheet",
158            filename=nome_anexo,
159        )
160
161        if caminho_planilha and not manter_planilhas:
162            caminho_planilha.unlink(missing_ok=True)
163
164        servidor_email.send_message(msg)
165        registrar_log(f"E-mail para {id_log} enviado com sucesso!", "INFO", exibir_prints)
166
167        return True
168    except Exception as erro:
169        registrar_log(f"Falha ao enviar e-mail para {id_log}: {erro}", "CRITICAL")
170        return False

173def _processar_pdf_e_enviar_email(
174    servidor_email: smtplib.SMTP,
175    assunto: str,
176    destinatario: str,
177    html_corpo: str,
178    dic_mapa_cids: dict[str, str],
179    dic_imagens_originais: dict[str, bytes],
180    caminho_pdf: Path | None,
181    eh_notebook: bool,
182    msg_loja: bool,
183    caminho_planilha: Path | None,
184    bytes_anexo: bytes,
185    nome_anexo_email: str,
186    nome_anexo_pdf: str,
187    manter_planilhas: bool,
188    exibir_prints: bool,
189    id_log: str,
190) -> bool:
191    """Mapeia os CIDs das imagens, gera o PDF opcional de simulação e envia o e-mail.
192
193    Args:
194        servidor_email (smtplib.SMTP): Servidor SMTP autenticado.
195        assunto (str): Assunto do e-mail.
196        destinatario (str): Endereço de e-mail do destinatário.
197        html_corpo (str): Corpo HTML renderizado.
198        dic_mapa_cids (dict[str, str]): Mapeamento {cid_original: cid_sem_colchetes}.
199        dic_imagens_originais (dict[str, bytes]): Mapeamento {cid_original: img_bytes}.
200        caminho_pdf (Path | None): Caminho para salvar o PDF de simulação, ou None.
201        eh_notebook (bool): Indicador de ambiente Jupyter Notebook.
202        msg_loja (bool): Indica se o layout do PDF deve ser ajustado para loja.
203        caminho_planilha (Path | None): Caminho físico da planilha da loja, se houver.
204        bytes_anexo (bytes): Conteúdo binário do anexo Excel.
205        nome_anexo_email (str): Nome do anexo enviado no e-mail.
206        nome_anexo_pdf (str): Nome do anexo exibido no cartão visual do PDF.
207        manter_planilhas (bool): Indica se a planilha física deve ser mantida após o envio.
208        exibir_prints (bool): Indica se mensagens informativas devem ser exibidas.
209        id_log (str): Identificador para registro de log.
210
211    Returns:
212        bool: True se o envio foi bem-sucedido, False caso contrário.
213    """
214    dic_cids_sem_colch = {dic_mapa_cids[chave]: img_bytes for chave, img_bytes in dic_imagens_originais.items()}
215    dic_cids_com_colch = {f"<{dic_mapa_cids[chave]}>": img_bytes for chave, img_bytes in dic_imagens_originais.items()}
216
217    if caminho_pdf:
218        gerar_pdf_do_email(
219            msg_loja=msg_loja,
220            html_email=html_corpo,
221            dic_imagens_embutidas=dic_cids_sem_colch,
222            eh_notebook=eh_notebook,
223            caminho_pdf=caminho_pdf,
224            exibir_prints=exibir_prints,
225            nome_anexo=nome_anexo_pdf,
226            bytes_anexo=bytes_anexo,
227        )
228
229    return enviar_email(
230        servidor_email=servidor_email,
231        assunto=assunto,
232        remetente=servidor_email.user,
233        destinatario=destinatario,
234        html_corpo=html_corpo,
235        dic_imagens_embutidas=dic_cids_com_colch,
236        caminho_planilha=caminho_planilha,
237        bytes_anexo=bytes_anexo,
238        nome_anexo=nome_anexo_email,
239        manter_planilhas=manter_planilhas,
240        exibir_prints=exibir_prints,
241        id_log=id_log,
242    )

245def _preparar_ranking_lojas(
246    df_consolidado: pd.DataFrame,
247    dic_metas_lojas: dict,
248    dic_nomes_lojas: dict[int, str],
249    sufixo_col: str,
250    sufixo_meta: str,
251) -> pd.DataFrame:
252    """Ordena, renomeia, adiciona metas e estrutura a tabela de ranking para um período específico.
253
254    Args:
255        df_consolidado (pd.DataFrame): Tabela consolidada das lojas.
256        dic_metas_lojas (dict): Dicionário com as metas das lojas.
257        dic_nomes_lojas (dict[int, str]): Dicionário mapeando {id_loja: nome_loja}.
258        sufixo_col (str): Sufixo das colunas do DataFrame ('Ano' ou 'Ontem').
259        sufixo_meta (str): Sufixo das chaves no dicionário de metas ('ano' ou 'dia').
260
261    Returns:
262        pd.DataFrame: Tabela de ranking pronta para exportação.
263    """
264    col_fat = f"Fat_{sufixo_col}"
265    df_rank = df_consolidado.sort_index(ascending=True).sort_values(by=col_fat, ascending=False).copy()
266
267    df_rank = df_rank.rename(
268        columns={
269            f"Fat_{sufixo_col}": "Faturamento (R$)",
270            f"Meta_Fat_{sufixo_col}": "Faturamento Ok",
271            f"Prod_Div_{sufixo_col}": "Diversidade de Produtos",
272            f"Meta_Div_{sufixo_col}": "Diversidade Ok",
273            f"Tic_Medio_{sufixo_col}": "Ticket Médio (R$)",
274            f"Meta_Tic_{sufixo_col}": "Ticket Ok",
275        }
276    )
277
278    df_rank["Loja"] = df_rank.index.astype(str) + " - " + df_rank.index.map(dic_nomes_lojas)
279    df_rank["Meta Faturamento (R$)"] = dic_metas_lojas[f"fat_{sufixo_meta}"]
280    df_rank["Meta Diversidade"] = dic_metas_lojas[f"div_{sufixo_meta}"]
281    df_rank["Meta Ticket (R$)"] = dic_metas_lojas[f"tic_{sufixo_meta}"]
282
283    return df_rank[
284        [
285            "Loja",
286            "Faturamento (R$)",
287            "Meta Faturamento (R$)",
288            "Faturamento Ok",
289            "Diversidade de Produtos",
290            "Meta Diversidade",
291            "Diversidade Ok",
292            "Ticket Médio (R$)",
293            "Meta Ticket (R$)",
294            "Ticket Ok",
295        ]
296    ]

299def enviar_emails_lojas(
300    servidor_email: smtplib.SMTP,
301    nome_remetente: str,
302    df_lojas: pd.DataFrame,
303    dic_metas: dict,
304    df_vendas_consolidadas_lojas: pd.DataFrame,
305    data_ontem_str_dma: str,
306    data_ontem_str_amd: str,
307    template_compilado: Template,
308    dic_planilhas_lojas: dict[int, Path],
309    caminho_graficos: Path | None,
310    caminho_pdfs: Path | None,
311    eh_notebook: bool,
312    exibir_prints: bool,
313    manter_planilhas: bool,
314) -> bool:
315    """Envia e-mails individuais às lojas com desempenho diário/anual, gráfico mensal e planilha de vendas.
316
317    Args:
318        servidor_email (smtplib.SMTP): Servidor de e-mail ativo.
319        nome_remetente (str): Nome do remetente.
320        df_lojas (pd.DataFrame): Tabela de dados cadastrais das lojas.
321        dic_metas (dict): Dicionário com as metas diárias e anuais das lojas.
322        df_vendas_consolidadas_lojas (pd.DataFrame): Tabela de vendas consolidadas das lojas.
323        data_ontem_str_dma (str): Data do dia anterior em formato dd/mm/aaaa.
324        data_ontem_str_amd (str): Data do dia anterior em formato aaaa-mm-dd.
325        template_compilado (Template): Template HTML compilado das lojas.
326        dic_planilhas_lojas (dict[int, Path]): Dicionário mapeando ID da loja ao caminho da sua planilha.
327        caminho_graficos (Path | None): Diretório para salvar imagens dos gráficos ou None.
328        caminho_pdfs (Path | None): Diretório para salvar PDFs de simulação ou None.
329        eh_notebook (bool): Indicador de ambiente Jupyter Notebook.
330        exibir_prints (bool): Indicador de exibição de mensagens informativas.
331        manter_planilhas (bool): Indicador de manutenção das planilhas em disco após o envio.
332
333    Returns:
334        bool: Sucesso ou falha no envio para todas as lojas.
335    """
336    try:
337        df_lojas_geral = df_vendas_consolidadas_lojas.join(df_lojas[["Loja", "Gerente", "E-mail"]])
338
339        for id_loja, colunas in df_lojas_geral.iterrows():
340            id_log = f"loja {id_loja}"
341            loja_id_nome = f"{id_loja} - {colunas['Loja']}"
342            registrar_log(f"Processando e-mail da loja: {loja_id_nome}", "INFO", exibir_prints)
343
344            caminho_planilha_loja = dic_planilhas_lojas[id_loja]
345            df_mensal_loja = pd.read_excel(caminho_planilha_loja, sheet_name="Mensal")
346
347            html_loja, dic_cids_loja = construir_html_email(
348                nome_remetente=nome_remetente,
349                data_ontem_str_dma=data_ontem_str_dma,
350                colunas_dados=colunas,
351                dic_metas=dic_metas,
352                template=template_compilado,
353                lst_cids_imagens=["grafico_vendas"],
354                id_log=id_log,
355                gerente=colunas["Gerente"],
356                nome_loja=colunas["Loja"],
357            )
358            if html_loja is None:
359                return False
360
361            caminho_img_loja = (
362                caminho_graficos / f"{data_ontem_str_amd}_Vendas_Mensais_Loja_{id_loja}.png"
363                if caminho_graficos
364                else None
365            )
366            img_bytes = construir_grafico_evol_vendas(
367                loja_id_nome=loja_id_nome,
368                meses=df_mensal_loja["Mês"],
369                quantidades=df_mensal_loja["Quantidade"],
370                faturamentos=df_mensal_loja["Valor Final"],
371                caminho_grafico=caminho_img_loja,
372                id_log=id_log,
373            )
374            if not img_bytes:
375                return False
376
377            bytes_planilha = caminho_planilha_loja.read_bytes()
378            caminho_pdf_loja = (
379                caminho_pdfs / f"{data_ontem_str_amd} - OnePage - Loja {loja_id_nome}.pdf"
380                if caminho_pdfs
381                else None
382            )
383
384            if not _processar_pdf_e_enviar_email(
385                servidor_email=servidor_email,
386                assunto=f"OnePage Dia {data_ontem_str_dma} - Loja {loja_id_nome}",
387                destinatario=colunas["E-mail"],
388                html_corpo=html_loja,
389                dic_mapa_cids=dic_cids_loja,
390                dic_imagens_originais={"grafico_vendas": img_bytes},
391                caminho_pdf=caminho_pdf_loja,
392                eh_notebook=eh_notebook,
393                msg_loja=True,
394                caminho_planilha=caminho_planilha_loja,
395                bytes_anexo=bytes_planilha,
396                nome_anexo_email=caminho_planilha_loja.name,
397                nome_anexo_pdf=caminho_planilha_loja.name,
398                manter_planilhas=manter_planilhas,
399                exibir_prints=exibir_prints,
400                id_log=id_log,
401            ):
402                return False
403
404        return True
405    except Exception as erro:
406        registrar_log(f"Erro no envio de e-mails às lojas: {erro}!", "CRITICAL")
407        return False

410def enviar_email_diretoria(
411    servidor_email: smtplib.SMTP,
412    nome_remetente: str,
413    df_vendas: pd.DataFrame,
414    dic_metas_lojas: dict,
415    dic_metas_rede: dict,
416    df_vendas_consolidadas_lojas: pd.DataFrame,

423    caminho_pdfs: Path | None,
424    eh_notebook: bool,
425    dic_nomes_lojas: dict[int, str],
426    exibir_prints: bool,
427) -> bool:
428    """Envia e-mail consolidado à diretoria com indicadores da rede, gráficos e planilha de rankings.
429
430    Args:
431        servidor_email (smtplib.SMTP): Servidor de e-mail ativo.
432        nome_remetente (str): Nome do remetente.
433        df_vendas (pd.DataFrame): Tabela geral de vendas.

435        dic_metas_rede (dict): Dicionário com as metas anuais e diárias da rede.
436        df_vendas_consolidadas_lojas (pd.DataFrame): Tabela de vendas consolidadas das lojas.
437        df_vendas_consolidadas_rede (pd.DataFrame): Tabela de vendas consolidadas de toda a rede.
438        email_diretoria (str): Endereço de e-mail da diretoria.
439        data_ontem_str_dma (str): Data do dia anterior em formato dd/mm/aaaa.
440        data_ontem_str_amd (str): Data do dia anterior em formato aaaa-mm-dd.

442        caminho_graficos (Path | None): Diretório para salvar imagens dos gráficos ou None.
443        caminho_pdfs (Path | None): Diretório para salvar PDFs de simulação ou None.
444        eh_notebook (bool): Indicador de ambiente Jupyter Notebook.
445        dic_nomes_lojas (dict[int, str]): Dicionário de nomes das lojas.
446        exibir_prints (bool): Indicador de exibição de mensagens informativas.
447
448    Returns:
449        bool: Sucesso ou falha no envio à diretoria.
450    """

452        df_ranking_ano = _preparar_ranking_lojas(
453            df_vendas_consolidadas_lojas, dic_metas_lojas, dic_nomes_lojas, "Ano", "ano"
454        )
455        df_ranking_ontem = _preparar_ranking_lojas(
456            df_vendas_consolidadas_lojas, dic_metas_lojas, dic_nomes_lojas, "Ontem", "dia"
457        )
458
459        fat_ano_max = df_ranking_ano["Faturamento (R$)"].iloc[0]
460        lojas_max_ano = df_ranking_ano[df_ranking_ano["Faturamento (R$)"] == fat_ano_max]["Loja"].tolist()
461        fat_ano_min = df_ranking_ano["Faturamento (R$)"].iloc[-1]
462        lojas_min_ano = df_ranking_ano[df_ranking_ano["Faturamento (R$)"] == fat_ano_min]["Loja"].tolist()
463
464        fat_ontem_max = df_ranking_ontem["Faturamento (R$)"].iloc[0]
465        lojas_max_ontem = df_ranking_ontem[df_ranking_ontem["Faturamento (R$)"] == fat_ontem_max]["Loja"].tolist()
466        fat_ontem_min = df_ranking_ontem["Faturamento (R$)"].iloc[-1]
467        lojas_min_ontem = df_ranking_ontem[df_ranking_ontem["Faturamento (R$)"] == fat_ontem_min]["Loja"].tolist()
468
469        id_log = "diretoria"
470
471        html_diretoria, dic_cids_diretoria = construir_html_email(
472            nome_remetente=nome_remetente,
473            data_ontem_str_dma=data_ontem_str_dma,
474            colunas_dados=df_vendas_consolidadas_rede.iloc[0],
475            dic_metas=dic_metas_rede,
476            template=template_compilado,
477            lst_cids_imagens=["grafico_mensal_rede", "grafico_anual_lojas"],
478            id_log=id_log,
479            qtd_lojas=len(df_vendas_consolidadas_lojas),
480            fat_ano_max=converter_notacao_europeia(fat_ano_max),
